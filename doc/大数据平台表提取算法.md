```python
class NQIAuth:
    def login(self, username, password, captcha_code):
        """执行登录"""
        # 1. 构造登录数据
        login_data = {
            'username': username,
            'password': self._encrypt_password(password),
            'captcha': captcha_code,
            'rememberMe': 'true'
        }
        
        # 2. 发送登录请求
        response = self.session.post(
            LOGIN_URL,
            data=login_data,
            headers=HEADERS,
            allow_redirects=False
        )
        
        # 3. 验证登录结果
        if self._verify_response(response):
            # 4. 保存CASTGC Cookie用于后续API调用
            self.castgc = self.session.cookies.get('CASTGC')
            return True
        return False
```

### 2.3 Cookie管理

登录成功后，系统会获取`CASTGC` Cookie，这是后续所有即席查询操作的身份凭证：

```python
# 从Cookie中安全获取CASTGC
def get_cookie_value(cookie_jar, name, domain=None):
    """处理多个同名cookie的情况"""
    try:
        value = cookie_jar.get(name, domain=domain)
        if value:
            return value
        return cookie_jar.get(name)
    except requests.cookies.CookieConflictError:
        # 存在多个同名cookie，手动遍历获取第一个
        cookies = [c for c in cookie_jar if c.name == name]
        if cookies:
            if domain:
                for c in cookies:
                    if c.domain == domain or (domain in c.domain):
                        return c.value
            return cookies[0].value
        return None
```

```python
# core/query.py - JXCXQuery.enter_jxcx()
def enter_jxcx(self, retry_times=None, timeout=None):
    """进入即席查询模块"""
    # 1. 获取CASTGC token
    castgc = get_cookie_value(self.sess.cookies, 'CASTGC', domain='nqi.gmcc.net')
    
    # 2. 构造请求URL
    url = f'{BASE_URL}/pro-portal/pure/urlAction.action'
    params = {
        'url': 'pro-adhoc/index',
        'random': random.random(),
        '__PID': 'JXCX',
        'token': castgc
    }
    
    # 3. 发送请求（带重试机制）
    for attempt in range(retry_times):
        res = self.sess.get(url, params=params, timeout=timeout)
        
        if res.status_code == 200:
            # 检查是否包含有效的adhoc内容
            if 'adhoc' in res.text or 'jxcx' in res.text.lower():
                self.enabled = True
                return True
        
        # 检查JSESSIONID是否更新
        jsessionid = get_cookie_value(self.sess.cookies, 'JSESSIONID')
        if jsessionid:
            self.enabled = True
            return True
    
    return False
```

```python
def _get_field_config_by_table(self, table_name):
```

### 4.2 字段配置缓存机制

```python
class JXCXQuery:
    def __init__(self, session):
        self.sess = session
        self._field_config_cache = {}  # 缓存字段配置
    
    def get_field_config(self, table_key, fieldtype, api_type='search', table_name=None):
        """获取字段配置（带缓存）"""
        # 生成缓存键
        cache_key = f"{table_key}_{fieldtype}_{api_type}"
        
        # 检查缓存
        if cache_key in self._field_config_cache:
            return self._field_config_cache[cache_key]
        
        # 从API获取
        if api_type == 'table':
            configs = self._get_field_config_by_table(table_name)
        else:
            configs = self._get_field_config_by_search(table_key, fieldtype)
        
        # 存入缓存
        self._field_config_cache[cache_key] = configs
        return configs
```

### 5.2 datatype 字段数据类型详解

`datatype` 字段用于指定字段的数据类型，对应数据库中的数据类型。

#### 5.2.1 datatype 取值对照表

| datatype 值 | 数据类型 | 说明 | 典型字段示例 |
|------------|---------|------|-------------|
| `character varying` | 字符串类型 | 可变长度字符串，默认类型 | `city`(地市)、`cgi`(小区标识)、`cell_name`(小区名)、`freq`(频段)、`cover_type`(覆盖类型)、`area`(区县)、`grid`(网格) |
| `timestamp` | 时间戳类型 | 日期时间类型，用于时间字段 | `starttime`(开始时间)、`endtime`(结束时间)、`busy_hour`(忙时) |
| `numeric` | 数值类型 | 整数或小数，用于计数/比率字段 | `volte_ul_tunzi_len`(VoLTE语音上行吞字时长)、`volte_ans_voice_call`(通话次数)、`cs_reg_suss_rate`(注册成功率) |
| `decimal` | 十进制小数 | 高精度小数，用于经纬度/百分比 | `longitude`(经度)、`latitude`(纬度)、`bh_pdcchcceoccupancyrate`(CCE占用率)、`upoctudl`(流量) |
| `boolean` | 布尔类型 | 是/否类型，用于标识字段 | `is_remote`(是否拉远)、`is_highload`(是否高负荷)、`is_interfere`(是否干扰) |
| `1` | 简化时间类型 | 特殊值，用于部分报表的时间维度字段 | 4G干扰报表的 `starttime`、`endtime` |

### 5.4 查询条件构建算法

#### 条件格式转换

系统支持两种条件格式，并自动转换：

```python
def convert_where_conditions(conditions):
    """转换where条件格式为API要求的格式
    
    新格式 -> 旧格式转换
    """
    result = []
    for i, cond in enumerate(conditions):
        # 新格式: {'field': 'xxx', 'operator': '>=', 'value': 'xxx'}
        if 'field' in cond and 'operator' in cond and 'value' in cond:
            field = cond['field']
            operator = cond['operator']
            value = cond['value']
            
            # 自动推断数据类型
            datatype = 'character'
            if 'time' in field.lower() or 'date' in field.lower():
                datatype = 'timestamp'
                # 时间格式处理：使用空格分隔符
                if '+' in value:
                    value = value.replace('+', ' ')
                elif ' ' not in value:
                    # 纯日期格式，需要添加时间部分
                    if operator in ('>=', '>'):
                        value = value + ' 00:00:00'
                    elif operator in ('<=', '<', '='):
                        value = value + ' 23:59:59'
            
            # 结束时间使用 < 而非 <=
            if operator == '<=' and datatype == 'timestamp':
                operator = '<'
            
            result.append({
                'datatype': datatype,
                'feild': field,
                'feildName': '',
                'symbol': operator,
                'val': value,
                'whereCon': 'and',
                'query': True
            })
    
    return result
```

### 5.5 Payload编码算法

```python
def _encode_payload(self, payload):
    """URL编码payload - 使用DataTables标准格式"""
    from urllib.parse import quote
    
    out_list = []
    for key in payload:
        if key == 'columns':
            # DataTables标准格式: columns[0][data]=field&columns[0][name]=&...
            col_parts = []
            for i, col in enumerate(payload[key]):
                for sub_key, sub_val in col.items():
                    if isinstance(sub_val, dict):
                        for ss_key, ss_val in sub_val.items():
                            col_parts.append(f'columns[{i}][{sub_key}][{ss_key}]={quote(str(ss_val))}')
                    else:
                        col_parts.append(f'columns[{i}][{sub_key}]={quote(str(sub_val))}')
            out_list.append('&'.join(col_parts))
        
        elif key == 'order':
            # DataTables标准格式: order[0][column]=0&order[0][dir]=desc
            order_parts = []
            for i, ord_item in enumerate(payload[key]):
                for sub_key, sub_val in ord_item.items():
                    order_parts.append(f'order[{i}][{sub_key}]={quote(str(sub_val))}')
            out_list.append('&'.join(order_parts))
        
        elif key == 'search':
            # DataTables标准格式: search[value]=&search[regex]=false
            search_parts = []
            for sub_key, sub_val in payload[key].items():
                search_parts.append(f'search[{sub_key}]={quote(str(sub_val))}')
            out_list.append('&'.join(search_parts))
        
        elif key in ['result', 'where']:
            # 使用JSON序列化
            json_str = json.dumps(payload[key], ensure_ascii=False, separators=(',', ':'))
            out_list.append(quote(key) + '=' + quote(json_str))
        
        else:
            out_list.append(quote(key) + '=' + quote(str(payload[key])))
    
    return '&'.join(out_list)
```

## 6. 数据查询算法

### 6.1 获取数据总数

```python
def get_table_count(self, payload, report_name=None):
    """获取查询结果行数"""
    # 筛选必要参数（不含columns/order/search等）
    key_list = [
        'geographicdimension', 'timedimension', 'enodebField', 'cgiField',
        'timeField', 'cellField', 'cityField', 'result', 'where', 'indexcount'
    ]
    count_payload = {key: value for key, value in payload.items() if key in key_list}
    
    # URL编码
    payload_encoded = self._encode_payload(count_payload)
    
    # 发送请求
    res = self.sess.post(
        JXCX_COUNT_URL,
        data=payload_encoded,
        headers=HEADERS,
        timeout=TIMEOUT_EXTRA_LONG  # 180秒超时
    )
    
    # 解析响应
    result = json.loads(res.content)
    count = self._extract_count_from_response(result)
    
    return count if count else 0

def _extract_count_from_response(self, result):
    """从API响应中提取count字段（支持多种响应格式）"""
    # 位置1: result['count']
    if 'count' in result:
        return result['count']
    
    # 位置2: result['data']['count']
    if 'data' in result and isinstance(result['data'], dict):
        if 'count' in result['data']:
            return result['data']['count']
    
    # 位置3: result['data']['total']
    if 'data' in result and isinstance(result['data'], dict):
        if 'total' in result['data']:
            return result['data']['total']
    
    # 位置4: result['recordsTotal'] (DataTables标准)
    if 'recordsTotal' in result:
        return result['recordsTotal']
    
    # 位置5: result['recordsFiltered']
    if 'recordsFiltered' in result:
        return result['recordsFiltered']
    
    return None
```

### 6.2 获取数据内容（一次性获取）

```python
def _fetch_by_loop(self, payload, total_count, progress_callback=None):
    """一次性获取全部数据（不分页，防止重复/遗漏）"""
    import copy
    
    # 构造请求参数
    p = copy.deepcopy(payload)
    p['start'] = 0
    p['length'] = total_count  # 一次性请求全部
    
    # 根据数据量动态设置超时
    timeout = max(60, min(600, total_count // 1000 * 3))
    
    if progress_callback:
        progress_callback(0, total_count, f"正在获取 {total_count} 条数据...")
    
    # 发送数据请求
    data_list = self._fetch_data(p, timeout=timeout)
    
    if data_list:
        if progress_callback:
            progress_callback(len(data_list), total_count, f"获取完成: {len(data_list)} 条")
        return data_list
    
    return []
```

### 6.3 完整查询流程

```python
def get_table(self, payload, to_df=True, progress_callback=None, report_name="未知报表"):
    """获取表格数据（完整流程）"""
    import pandas as pd
    
    # Step 1: 检查Session有效性
    if not self.check_session_valid():
        if not self.enter_jxcx(retry_times=2, timeout=60):
            return pd.DataFrame()
    
    # Step 2: 获取数据总数
    count_payload = {key: value for key, value in payload.items() 
                   if key in ['geographicdimension', 'timedimension', 'enodebField', 
                             'cgiField', 'timeField', 'cellField', 'cityField', 
                             'result', 'where', 'indexcount']}
    
    total_count = self.get_table_count(count_payload, report_name=report_name)
    
    if total_count == 0:
        return pd.DataFrame()
    
    # Step 3: 获取数据内容
    data_payload = {key: value for key, value in payload.items()
                   if key in ['geographicdimension', 'timedimension', 'enodebField',
                             'cgiField', 'timeField', 'cellField', 'cityField',
                             'result', 'where', 'indexcount']}
    
    data_list = self._fetch_by_loop(data_payload, total_count, progress_callback)
    
    # Step 4: 转换为DataFrame
    if to_df:
        res_df = pd.DataFrame(data_list) if data_list else pd.DataFrame()
        
        # Step 5: 应用字段映射（英文字段名 → 中文字段名）
        en_zh_df = self._get_field_mapping(payload)
        if not en_zh_df.empty:
            res_df = pd.concat([en_zh_df, res_df], ignore_index=True)
            index_first = res_df.index.tolist()[0]
            to_colname = list(res_df.loc[index_first])
            res_df.columns = to_colname
            res_df.drop(index=index_first, inplace=True)
        
        return res_df
    else:
        return {'data': data_list}
```

```python
def export_dataframe_streaming(df, filepath, sheet_name='Sheet1',
                               header_color=None, batch_size=None,
                               progress_callback=None):
    """流式导出DataFrame到Excel（适合超大数据）
    
    分批写入，每批格式化一次，显著减少内存占用
    """
    if batch_size is None:
        batch_size = EXCEL_BATCH_SIZE  # 默认5000行/批
    
    wb = Workbook()
    ws = wb.active
    ws.title = sheet_name
    
    # 写入表头
    header = list(df.columns)
    fill, font, alignment, border = ExcelStyler.get_header_style('header_blue')
    
    for col_idx, col_name in enumerate(header, start=1):
        cell = ws.cell(row=1, column=col_idx, value=col_name)
        cell.fill = fill
        cell.font = font
        cell.alignment = alignment
        cell.border = border
    
    # 分批写入数据
    total_rows = len(df)
    for batch_start in range(0, total_rows, batch_size):
        batch_end = min(batch_start + batch_size, total_rows)
        batch_df = df.iloc[batch_start:batch_end]
        
        for row_idx, (_, row) in enumerate(batch_df.iterrows(), 
                                          start=batch_start + 2):
            for col_idx, value in enumerate(row, start=1):
                try:
                    ws.cell(row=row_idx, column=col_idx, value=value)
                except (ValueError, TypeError):
                    pass
        
        if progress_callback:
            progress_callback(batch_end, total_rows,
                            f"正在写入... {batch_end}/{total_rows} 行")
    
    wb.save(filepath)
    return True
```

## 9. 并行查询算法

### 9.1 多报表并行查询

```python
def query_tables_parallel(self, table_configs, max_workers=None, 
                         progress_callback=None):
    """并行查询多个报表
    
    Args:
        table_configs: 表配置列表 [(table_name, payload), ...]
        max_workers: 最大并行数（默认3）
    """
    if max_workers is None:
        max_workers = MAX_PARALLEL_QUERIES  # 默认3
    
    results = {}
    completed = 0
    total = len(table_configs)
    
    def query_single_table(table_name, payload):
        """查询单个表（线程池任务）"""
        try:
            df = self.get_table(payload, to_df=True, report_name=table_name)
            return table_name, df
        except Exception as e:
            return table_name, pd.DataFrame()
    
    # 使用线程池并行查询
    with ThreadPoolExecutor(max_workers=max_workers) as executor:
        futures = {
            executor.submit(query_single_table, name, payload): name
            for name, payload in table_configs
        }
        
        for future in as_completed(futures):
            table_name = futures[future]
            completed += 1
            
            try:
                name, df = future.result()
                results[name] = df
                
                if progress_callback:
                    status = "成功" if not df.empty else "空数据"
                    progress_callback(completed, total, name, status)
            except Exception as e:
                results[table_name] = pd.DataFrame()
    
    return results
```
